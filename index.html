import { doc, getDoc, updateDoc } from "firebase/firestore"; // एडमिन या पेमेंट सक्सेस होने पर कॉल करें: days = 30, 60, या 90 async function updateDatabaseVIP(db, userId, daysToAdd, planPrice) { const userRef = doc(db, "users", userId); const userSnap = await getDoc(userRef); const now = Date.now(); let currentExpiry = 0; if (userSnap.exists() && userSnap.data().vipExpiry) { currentExpiry = userSnap.data().vipExpiry; } // अगर पुराना प्लान चल रहा है तो उसी में आगे जोड़ें, वरना आज से जोड़ें const baseTime = (currentExpiry > now) ? currentExpiry : now; const newExpiry = baseTime + (daysToAdd * 24 * 60 * 60 * 1000); await updateDoc(userRef, { isVIP: true, vipExpiry: newExpiry, lastPlanPrice: planPrice, lastUpdated: now }); console.log(`User ${userId} VIP extended by ${daysToAdd} days!`); }
